### **Modo Producción (con resiliencia completa)**

```python
from tables.connectors import KrakenConnector
from core.testing_session_resilient import ResilientSessionHelper

# Conector base
kraken = KrakenConnector(api_key, secret, testnet=False)

# Setup resiliencia
helper = ResilientSessionHelper(
    connector=kraken,
    session_id="paroli_BTC_5m_20251104",
    player_name="paroli",
    symbol="BTC/USD",
    timeframe="5m",
    state_dir="./state/production"
)

# Get resilient connector
resilient_connector = helper.get_resilient_connector()

# Try recover previous session
recovered_state = await helper.try_recover()
if recovered_state:
    print(f"Recuperando sesión con {recovered_state['candles_processed']} velas")

# Create table
table = TableCCXTPro(connector=resilient_connector, symbol="BTC/USD")

await table.connect()
# Sistema robusto listo para producción 24/7
```
